def num(v, default=None):
    """'85%', '32 C', '5460', 'N/A' -> float"""
    if v is None:
        return default
    if isinstance(v, bool):
        return float(v)
    if isinstance(v, (int, float)):
        return float(v)
    s = str(v).strip().rstrip("%").replace(",", "").split(" ")[0]
    try:
        return float(s)
    except ValueError:
        return default

class Metrics:
    """Gom metric của 1 target. Label 'target','vendor' tự thêm vào đầu."""

    DEFS = {
        # name: (type, help, extra labels)
        "up": ("g", "1 nếu exporter đăng nhập & lấy dữ liệu được", []),
        "scrape_duration_seconds": ("g", "Thời gian scrape 1 target", []),
        "section_error": ("g", "1 nếu lệnh API của section lỗi", ["section"]),
        "info": ("g", "Thông tin hệ thống", ["name", "model", "serial", "version", "ip"]),
        "system_health": ("g", "Health tổng (0=OK,1=Warning,2=Critical,3=Unknown)", []),
        "alerts_active": ("g", "Số alert đang active/chưa fix theo mức độ", ["severity"]),
        "alert_info": ("g", "Chi tiết alert đang active (value=1)",
                       ["severity", "component", "message", "acknowledged"]),
        "component_health": ("g", "Health linh kiện (0=OK,1=Warning,2=Critical,3=Unknown)",
                             ["type", "component", "status"]),
        "disk_temperature_celsius": ("g", "Nhiệt độ disk", ["component"]),
        "disk_ssd_life_left_percent": ("g", "SSD life left", ["component"]),
        "pool_health": ("g", "Health pool", ["pool"]),
        "pool_size_bytes": ("g", "Dung lượng pool", ["pool"]),
        "pool_free_bytes": ("g", "Dung lượng trống pool", ["pool"]),
        "pool_subscribed_bytes": ("g", "Dung lượng đã cấp phát cho volume (provisioned)", ["pool"]),
        "volume_health": ("g", "Health volume/LUN", ["volume", "pool"]),
        "volume_size_bytes": ("g", "Dung lượng volume/LUN", ["volume", "pool"]),
        "volume_allocated_bytes": ("g", "Dung lượng đã dùng volume/LUN", ["volume", "pool"]),
        "host_health": ("g", "Health host đã map (Unity/FlashSystem)", ["host", "status"]),
        "job_running": ("g", "Job đang chạy (rebuild/scrub/...)", ["job", "object"]),
        "cpu_percent": ("g", "CPU controller/SP/node", ["controller"]),
        "iops": ("g", "IOPS", ["controller", "op"]),
        "throughput_bytes_per_second": ("g", "Throughput", ["controller", "op"]),
        "latency_seconds": ("g", "Độ trễ trung bình", ["controller", "op"]),
        "cache_used_percent": ("g", "Write cache used", ["controller"]),
    }

    def __init__(self, target, vendor):
        self.target, self.vendor = target, vendor
        self.fams = {}

    def set(self, name, value, *labels):
        if value is None:
            return
        typ, help_, extra = self.DEFS[name]
        full = f"storage_{name}"
        if full not in self.fams:
            cls = GaugeMetricFamily if typ == "g" else CounterMetricFamily
            self.fams[full] = cls(full, help_, labels=["target", "vendor", *extra])
        self.fams[full].add_metric([self.target, self.vendor, *[str(x) for x in labels]], float(value))

    def families(self):
        return list(self.fams.values())

class Driver:
    vendor = "generic"
    sections = []  # list of method names

    def __init__(self, name, cfg):
        self.name = name
        self.host = cfg["host"]
        self.user = cfg["username"]
        self.password = cfg.get("password") or os.environ.get(cfg.get("password_env", ""), "")
        self.verify = cfg.get("verify_ssl", False)
        self.timeout = cfg.get("timeout", 30)
        self.port = cfg.get("port")
        self.s = requests.Session()
        self.s.verify = self.verify
        self.lock = threading.Lock()
        self.logged_in = False
        # alert "theo thiết kế" (vd mgmtport_b ở DR) -> không tính vào alerts_active
        self.ignore = [x.lower() for x in cfg.get("ignore_alerts", [])]

    def login(self):
        raise NotImplementedError

    def is_ignored(self, *texts):
        t = " ".join(str(x) for x in texts).lower()
        return any(i in t for i in self.ignore)

    def add_alert(self, m, counts, sev, component, message, acked):
        if sev not in ("critical", "warning"):
            return
        if self.is_ignored(component, message):
            m.set("alert_info", 1, sev, component, message[:200], "ignored")
            self.ignored_hits += 1
            return
        counts[sev] = counts.get(sev, 0) + 1
        m.set("alert_info", 1, sev, component, message[:200], acked)

    def _ensure_login(self, force=False):
        if force or not self.logged_in:
            self.s.cookies.clear()
            self.login()
            self.logged_in = True

    def collect(self, m: Metrics):
        """Chạy từng section. Section đầu (system) lỗi -> login lại & thử 1 lần nữa."""
        self.ignored_hits = 0
        with self.lock:
            for attempt in (1, 2):
                try:
                    self._ensure_login(force=attempt == 2)
                    getattr(self, self.sections[0])(m)
                    m.set("section_error", 0, self.sections[0])
                    break
                except Exception as e:
                    self.logged_in = False
                    if attempt == 2:
                        log.error("[%s] %s: %s", self.name, self.sections[0], e)
                        m.set("section_error", 1, self.sections[0])
                        return 0
            for sec in self.sections[1:]:
                try:
                    getattr(self, sec)(m)
                    m.set("section_error", 0, sec)
                except Exception as e:
                    m.set("section_error", 1, sec)
                    log.error("[%s] %s: %s", self.name, sec, e)
        return 1


def _alert_counts(m, counts):
    for sev in ("critical", "warning", "info"):
        m.set("alerts_active", counts.get(sev, 0), sev)

# Dell PowerVault ME4 / ME5

PV_HEALTH = {0: OK, 1: WARN, 2: CRIT, 3: UNKNOWN, 4: None}  # 4 = N/A -> bỏ qua
BLOCK = 512


class PowerVault(Driver):
    vendor = "powervault"
    sections = ["system", "alerts", "controllers", "ctrl_stats", "disks", "disk_groups",
                "pools", "volumes", "ports", "fans", "psus", "sensors"]

    @property
    def base(self):
        return f"https://{self.host}{':' + str(self.port) if self.port else ''}/api"

    def login(self):
        self.s.headers.update({"datatype": "json"})
        self.s.headers.pop("sessionKey", None)
        digest = hashlib.sha256(f"{self.user}_{self.password}".encode()).hexdigest()
        r = self.s.get(f"{self.base}/login/{digest}", timeout=self.timeout)
        key = self._key(r)
        if not key:  # ME5 firmware mới: basic auth
            r = self.s.get(f"{self.base}/login", auth=(self.user, self.password), timeout=self.timeout)
            key = self._key(r)
        if not key:
            raise RuntimeError(f"login failed HTTP {r.status_code}: {r.text[:200]}")
        self.s.headers["sessionKey"] = key

    @staticmethod
    def _key(r):
        try:
            st = r.json()["status"][0]
            if st.get("response-type-numeric") == 0 or st.get("return-code") == 1:
                return st.get("response")
        except Exception:
            pass
        return None

    def show(self, cmd):
        r = self.s.get(f"{self.base}/show/{cmd}", timeout=self.timeout)
        if r.status_code in (401, 403):
            raise RuntimeError(f"HTTP {r.status_code} (session hết hạn)")
        data = r.json()
        st = (data.get("status") or [{}])[0]
        if st.get("response-type") == "Error":
            raise RuntimeError(st.get("response"))
        return data

    @staticmethod
    def h(x):
        return PV_HEALTH.get(int(num(x.get("health-numeric"), 3)), UNKNOWN)

    # ---- sections ----
    def system(self, m):
        for s in self.show("system").get("system", []):
            m.set("info", 1, s.get("system-name", ""), s.get("product-id", ""),
                  s.get("midplane-serial-number", ""), "", self.host)
            m.set("system_health", self.h(s))

    def alerts(self, m):
        counts = {}
        for a in self.show("alerts").get("alerts", []):
            if str(a.get("resolved", "No")).lower() == "yes":
                continue
            sevn = str(a.get("severity", "")).lower()
            sev = "critical" if sevn in ("critical", "error") else "warning" if sevn == "warning" else "info"
            self.add_alert(m, counts, sev, a.get("component-id") or a.get("component", ""),
                           a.get("reason") or a.get("description") or "", str(a.get("acknowledged", "")))
        _alert_counts(m, counts)

    def controllers(self, m):
        for c in self.show("controllers").get("controllers", []):
            cid = c.get("durable-id", "")
            m.set("component_health", self.h(c), "controller", cid, c.get("status", ""))
            red = str(c.get("redundancy-status", "")).lower()
            m.set("component_health", OK if ("redundant" in red and "not redundant" not in red) else WARN,
                  "redundancy", cid, c.get("redundancy-status", ""))

    def ctrl_stats(self, m):
        for c in self.show("controller-statistics").get("controller-statistics", []):
            cid = c.get("durable-id", "")
            m.set("cpu_percent", num(c.get("cpu-load"), 0), cid)
            m.set("cache_used_percent", num(c.get("write-cache-used"), 0), cid)
            m.set("iops", num(c.get("iops"), 0), cid, "total")
            m.set("throughput_bytes_per_second", num(c.get("bytes-per-second-numeric"), 0), cid, "total")

    def disks(self, m):
        for d in self.show("disks").get("drives", []):
            loc = d.get("location", d.get("durable-id", ""))
            m.set("component_health", self.h(d), "disk", loc, d.get("usage", ""))
            m.set("disk_temperature_celsius", num(d.get("temperature-numeric"), num(d.get("temperature"))), loc)
            life = num(d.get("ssd-life-left-numeric"))
            if life is not None and life <= 100:
                m.set("disk_ssd_life_left_percent", life, loc)

    def disk_groups(self, m):
        for g in self.show("disk-groups").get("disk-groups", []):
            n = g.get("name", "")
            m.set("component_health", self.h(g), "diskgroup", n, g.get("status", ""))
            job = g.get("current-job", "")
            if job and job != "N/A":
                m.set("job_running", 1, job, n)

    def pools(self, m):
        for p in self.show("pools").get("pools", []):
            n = p.get("name", "")
            m.set("pool_health", self.h(p), n)
            m.set("pool_size_bytes", num(p.get("total-size-numeric"), 0) * BLOCK, n)
            m.set("pool_free_bytes", num(p.get("total-avail-numeric"), 0) * BLOCK, n)
            m.set("pool_subscribed_bytes", num(p.get("allocated-pages"), 0) * 4 * 1024 * 1024, n)

    def volumes(self, m):
        for v in self.show("volumes").get("volumes", []):
            n, pool = v.get("volume-name", ""), v.get("storage-pool-name", "")
            m.set("volume_health", self.h(v), n, pool)
            m.set("volume_size_bytes", num(v.get("size-numeric"), num(v.get("total-size-numeric"), 0)) * BLOCK, n, pool)
            m.set("volume_allocated_bytes", num(v.get("allocated-size-numeric"), 0) * BLOCK, n, pool)

    def ports(self, m):
        d = self.show("ports")
        for p in d.get("port", d.get("ports", [])):
            h = self.h(p)
            if h is not None:
                m.set("component_health", h, "host_port", p.get("port", ""), p.get("status", ""))

    def fans(self, m):
        d = self.show("fans")
        for f in d.get("fan", d.get("fans", [])):
            m.set("component_health", self.h(f), "fan", f.get("durable-id", f.get("name", "")), f.get("status", ""))

    def psus(self, m):
        for p in self.show("power-supplies").get("power-supplies", []):
            m.set("component_health", self.h(p), "psu", p.get("durable-id", p.get("name", "")), p.get("status", ""))

    def sensors(self, m):
        # status-numeric: 1=OK 2=Critical 3=Warning 4=Unrecoverable 5=Not installed 6=Unknown 7=Unavailable
        mp = {1: OK, 2: CRIT, 3: WARN, 4: CRIT, 6: UNKNOWN}
        for s in self.show("sensor-status").get("sensors", []):
            st = int(num(s.get("status-numeric"), 6))
            if st in mp:
                m.set("component_health", mp[st], "sensor", s.get("sensor-name", ""), s.get("status", ""))

def unity_health(v):
    # HealthEnum: 0 UNKNOWN, 5 OK, 7 OK_BUT, 10 DEGRADED, 15 MINOR, 20 MAJOR, 25 CRITICAL, 30 NON_RECOVERABLE
    v = int(num(v, 0))
    if v in (5, 7):
        return OK
    if v in (10, 15):
        return WARN
    if v >= 20:
        return CRIT
    return UNKNOWN


class Unity(Driver):
    vendor = "unity"
    sections = ["system", "alerts", "components", "pools", "luns", "hosts", "jobs", "perf"]
    COMPONENTS = {  # type -> resource
        "sp": "storageProcessor", "disk": "disk", "dpe": "dpe", "dae": "dae", "lcc": "lcc",
        "psu": "powerSupply", "fan": "fan", "battery": "battery", "memory": "memoryModule",
        "ssd": "ssd", "io_module": "ioModule", "fc_port": "fcPort", "eth_port": "ethernetPort",
        "sas_port": "sasPort",
    }

    @property
    def base(self):
        return f"https://{self.host}{':' + str(self.port) if self.port else ''}/api"

    def login(self):
        self.s.headers.update({"X-EMC-REST-CLIENT": "true", "Accept": "application/json"})
        r = self.s.get(f"{self.base}/types/loginSessionInfo/instances",
                       auth=(self.user, self.password), timeout=self.timeout)
        if r.status_code != 200:
            raise RuntimeError(f"login failed HTTP {r.status_code}: {r.text[:200]}")

    def get(self, res, fields, flt=None):
        params = {"fields": fields, "per_page": 2000, "compact": "true"}
        if flt:
            params["filter"] = flt
        r = self.s.get(f"{self.base}/types/{res}/instances", params=params, timeout=self.timeout)
        if r.status_code == 401:
            raise RuntimeError("HTTP 401 (session hết hạn)")
        r.raise_for_status()
        return [e.get("content", e) for e in r.json().get("entries", [])]

    @staticmethod
    def hv(obj):
        h = obj.get("health") or {}
        return unity_health(h.get("value"))

    @staticmethod
    def hdesc(obj):
        h = obj.get("health") or {}
        d = h.get("descriptions") or []
        return (d[0] if d else "")[:120]

    def system(self, m):
        for s in self.get("system", "name,model,serialNumber,health"):
            ver = ""
            try:
                sw = self.get("installedSoftwareVersion", "version")
                ver = sw[0].get("version", "") if sw else ""
            except Exception:
                pass
            m.set("info", 1, s.get("name", ""), s.get("model", ""), s.get("serialNumber", ""), ver, self.host)
            m.set("system_health", self.hv(s))

    def alerts(self, m):
        # SeverityEnum: 0 EMERGENCY 1 ALERT 2 CRITICAL 3 ERROR 4 WARNING 5 NOTICE 6 INFO 7 DEBUG 8 OK
        counts = {}
        for a in self.get("alert", "severity,isAcknowledged,state,message,component",
                          'isAcknowledged eq false'):
            if int(num(a.get("state"), 0)) == 2:  # INACTIVE
                continue
            sv = int(num(a.get("severity"), 6))
            sev = "critical" if sv <= 3 else "warning" if sv == 4 else "info"
            comp = a.get("component") or {}
            self.add_alert(m, counts, sev, comp.get("id", "") if isinstance(comp, dict) else str(comp),
                           a.get("message") or "", "false")
        _alert_counts(m, counts)

    def components(self, m):
        for typ, res in self.COMPONENTS.items():
            try:
                items = self.get(res, "name,health")
            except Exception as e:
                log.debug("[%s] unity %s: %s", self.name, res, e)
                continue
            for it in items:
                m.set("component_health", self.hv(it), typ, it.get("name", it.get("id", "")), self.hdesc(it))

    def pools(self, m):
        for p in self.get("pool", "name,health,sizeTotal,sizeFree,sizeSubscribed"):
            n = p.get("name", "")
            m.set("pool_health", self.hv(p), n)
            m.set("pool_size_bytes", num(p.get("sizeTotal"), 0), n)
            m.set("pool_free_bytes", num(p.get("sizeFree"), 0), n)
            m.set("pool_subscribed_bytes", num(p.get("sizeSubscribed"), 0), n)

    def luns(self, m):
        pools = {p["id"]: p.get("name", "") for p in self.get("pool", "id,name")}
        for l in self.get("lun", "name,health,sizeTotal,sizeAllocated,pool"):
            n = l.get("name", "")
            pool = pools.get((l.get("pool") or {}).get("id", ""), "")
            m.set("volume_health", self.hv(l), n, pool)
            m.set("volume_size_bytes", num(l.get("sizeTotal"), 0), n, pool)
            m.set("volume_allocated_bytes", num(l.get("sizeAllocated"), 0), n, pool)

    def hosts(self, m):
        for h in self.get("host", "name,health"):
            m.set("host_health", self.hv(h), h.get("name", ""), self.hdesc(h))

    def jobs(self, m):
        # JobStateEnum: 1 QUEUED 2 RUNNING 3 SUSPENDED 4 COMPLETED 5 FAILED 6 ROLLING_BACK 7 COMPLETED_WITH_PROBLEMS
        for j in self.get("job", "description,state", "state eq 2"):
            m.set("job_running", 1, (j.get("description") or "")[:80], "")

    def perf(self, m):
        """Metric lịch sử của Unity (interval 60s). Best-effort: firmware cũ có thể thiếu path."""
        paths = {
            "sp.*.cpu.summary.utilization": ("cpu_percent", None, 1),
            "sp.*.storage.summary.readsRate": ("iops", "read", 1),
            "sp.*.storage.summary.writesRate": ("iops", "write", 1),
            "sp.*.storage.summary.readBytesRate": ("throughput_bytes_per_second", "read", 1),
            "sp.*.storage.summary.writeBytesRate": ("throughput_bytes_per_second", "write", 1),
        }
        for path, (metric, op, mult) in paths.items():
            r = self.s.get(f"{self.base}/types/metricValue/instances",
                           params={"filter": f'path EQ "{path}"', "per_page": 1, "compact": "true"},
                           timeout=self.timeout)
            if r.status_code != 200:
                continue
            entries = r.json().get("entries", [])
            if not entries:
                continue
            vals = entries[0].get("content", {}).get("values", {})
            for sp, v in vals.items():
                v = num(v)
                if v is None:
                    continue
                if op:
                    m.set(metric, v * mult, sp, op)
                else:
                    m.set(metric, v * mult, sp)

def ibm_status(s):
    s = str(s).lower()
    if s in ("online", "active", "ok", "on", "yes", "normal", "spare", "candidate", "member"):
        return OK
    if s in ("degraded", "excluded", "service"):
        return WARN
    if s in ("offline", "failed", "inactive_configured", "error", "off"):
        return CRIT
    return UNKNOWN


class FlashSystem(Driver):
    vendor = "flashsystem"
    sections = ["system", "alerts", "nodes", "enclosures", "drives", "pools",
                "vdisks", "hosts", "fc_ports", "perf"]

    @property
    def base(self):
        return f"https://{self.host}:{self.port or 7443}/rest"

    def login(self):
        self.s.headers.pop("X-Auth-Token", None)
        r = self.s.post(f"{self.base}/auth", timeout=self.timeout,
                        headers={"X-Auth-Username": self.user, "X-Auth-Password": self.password})
        if r.status_code != 200 or "token" not in r.text:
            raise RuntimeError(f"login failed HTTP {r.status_code}: {r.text[:200]}")
        self.s.headers["X-Auth-Token"] = r.json()["token"]

    def cmd(self, name, body=None):
        r = self.s.post(f"{self.base}/{name}", json=body or {}, timeout=self.timeout)
        if r.status_code in (401, 403):
            raise RuntimeError(f"HTTP {r.status_code} (token hết hạn)")
        r.raise_for_status()
        d = r.json() if r.text.strip() else []
        return d if isinstance(d, list) else [d]

    def system(self, m):
        s = self.cmd("lssystem")[0]
        m.set("info", 1, s.get("name", ""), s.get("product_name", ""), s.get("id", ""),
              s.get("code_level", "").split(" ")[0], self.host)

    def alerts(self, m):
        counts = {}
        for e in self.cmd("lseventlog", {"alert": "yes", "fixed": "no", "message": "no"}):
            nt = str(e.get("notification_type", "")).lower()
            sev = "critical" if nt == "error" else "warning" if nt == "warning" else "info"
            self.add_alert(m, counts, sev, e.get("object_name") or e.get("object_type", ""),
                           f'{e.get("error_code", "")} {e.get("description", "")}'.strip(), "false")
        _alert_counts(m, counts)

    def nodes(self, m):
        try:
            nodes = self.cmd("lsnodecanister")
        except Exception:
            nodes = self.cmd("lsnode")
        for n in nodes:
            m.set("component_health", ibm_status(n.get("status")), "node", n.get("name", ""), n.get("status", ""))

    def enclosures(self, m):
        for e in self.cmd("lsenclosure"):
            m.set("component_health", ibm_status(e.get("status")), "enclosure", f'enc{e.get("id")}', e.get("status", ""))
        for p in self.cmd("lsenclosurepsu"):
            m.set("component_health", ibm_status(p.get("status")), "psu",
                  f'enc{p.get("enclosure_id")}-psu{p.get("PSU_id")}', p.get("status", ""))
        try:
            for b in self.cmd("lsenclosurebattery"):
                m.set("component_health", ibm_status(b.get("status")), "battery",
                      f'enc{b.get("enclosure_id")}-bat{b.get("battery_id")}', b.get("status", ""))
        except Exception:
            pass
        try:
            for c in self.cmd("lsenclosurecanister"):
                m.set("component_health", ibm_status(c.get("status")), "canister",
                      f'enc{c.get("enclosure_id")}-can{c.get("canister_id")}', c.get("status", ""))
        except Exception:
            pass

    def drives(self, m):
        for d in self.cmd("lsdrive", {"bytes": True}):
            st = d.get("status", "")
            h = ibm_status(st)
            if str(d.get("use", "")).lower() == "failed":
                h = CRIT
            m.set("component_health", h, "disk",
                  f'enc{d.get("enclosure_id")}-slot{d.get("slot_id")}', f'{st}/{d.get("use", "")}')

    def pools(self, m):
        for p in self.cmd("lsmdiskgrp", {"bytes": True}):
            n = p.get("name", "")
            m.set("pool_health", ibm_status(p.get("status")), n)
            m.set("pool_size_bytes", num(p.get("capacity"), 0), n)
            m.set("pool_free_bytes", num(p.get("free_capacity"), 0), n)
            m.set("pool_subscribed_bytes", num(p.get("virtual_capacity"), 0), n)

    def vdisks(self, m):
        for v in self.cmd("lsvdisk", {"bytes": True}):
            n, pool = v.get("name", ""), v.get("mdisk_grp_name", "")
            m.set("volume_health", ibm_status(v.get("status")), n, pool)
            m.set("volume_size_bytes", num(v.get("capacity"), 0), n, pool)

    def hosts(self, m):
        for h in self.cmd("lshost"):
            m.set("host_health", ibm_status(h.get("status")), h.get("name", ""), h.get("status", ""))

    def fc_ports(self, m):
        for p in self.cmd("lsportfc"):
            st = str(p.get("status", "")).lower()
            if st == "inactive_unconfigured":  # port không dùng
                continue
            m.set("component_health", ibm_status(st), "host_port",
                  f'{p.get("node_name", "")}-p{p.get("port_id", "")}', st)

    def perf(self, m):
        stats = {s.get("stat_name"): num(s.get("stat_current")) for s in self.cmd("lssystemstats")}
        ctl = "system"
        m.set("cpu_percent", stats.get("cpu_pc"), ctl)
        m.set("cache_used_percent", stats.get("write_cache_pc"), ctl)
        for op, pre in (("read", "vdisk_r_"), ("write", "vdisk_w_"), ("total", "vdisk_")):
            io, mb, ms = stats.get(pre + "io"), stats.get(pre + "mb"), stats.get(pre + "ms")
            m.set("iops", io, ctl, op)
            if mb is not None:
                m.set("throughput_bytes_per_second", mb * 1024 * 1024, ctl, op)
            if ms is not None:
                m.set("latency_seconds", ms / 1000.0, ctl, op)


DRIVERS = {"powervault": PowerVault, "me4": PowerVault, "me5": PowerVault,
           "unity": Unity, "flashsystem": FlashSystem, "storwize": FlashSystem}


# ===========================================================================
# Collector + HTTP
# ===========================================================================
class StorageCollector:
    def __init__(self, drivers, workers=5):
        self.drivers = drivers
        self.workers = workers

    def _one(self, name):
        d = self.drivers[name]
        m = Metrics(name, d.vendor)
        t0 = time.time()
        up = d.collect(m)
        m.set("up", up)
        if up and "storage_system_health" not in m.fams:
            # driver không có health tổng (FlashSystem) -> lấy mức xấu nhất của alert + linh kiện + pool
            worst = OK
            for fam in ("storage_component_health", "storage_pool_health"):
                for smp in getattr(m.fams.get(fam), "samples", []):
                    if smp.value in (WARN, CRIT):
                        worst = max(worst, int(smp.value))
            for smp in getattr(m.fams.get("storage_alerts_active"), "samples", []):
                if smp.value > 0:
                    worst = max(worst, CRIT if smp.labels["severity"] == "critical" else WARN)
            m.set("system_health", worst)
        # health Degraded chỉ do alert đã ignore (thiết kế) -> coi như OK
        sh, aa = m.fams.get("storage_system_health"), m.fams.get("storage_alerts_active")
        if up and sh and aa and getattr(d, "ignored_hits", 0) and \
                sum(x.value for x in aa.samples) == 0 and sh.samples[0].value == WARN:
            sh.samples[0] = sh.samples[0]._replace(value=float(OK))
        m.set("scrape_duration_seconds", time.time() - t0)
        return m.families()

    def collect(self, only=None):
        names = [only] if only else list(self.drivers)
        with ThreadPoolExecutor(max_workers=self.workers) as ex:
            results = list(ex.map(self._one, names))
        merged = {}
        for fams in results:
            for f in fams:
                if f.name in merged:
                    merged[f.name].samples.extend(f.samples)
                else:
                    merged[f.name] = f
        return list(merged.values())


class _Reg:
    def __init__(self, c, t):
        self.c, self.t = c, t

    def collect(self):
        return self.c.collect(self.t)


def render(collector, target=None):
    reg = CollectorRegistry(auto_describe=False)
    reg.register(_Reg(collector, target))
    return generate_latest(reg)


def make_handler(collector):
    class H(BaseHTTPRequestHandler):
        def do_GET(self):
            u = urlparse(self.path)
            if u.path in ("/metrics", "/probe"):
                t = parse_qs(u.query).get("target", [None])[0]
                if t and t not in collector.drivers:
                    return self._send(400, f"unknown target {t}\n".encode(), "text/plain")
                return self._send(200, render(collector, t), CONTENT_TYPE_LATEST)
            if u.path == "/":
                body = "<h3>Storage exporter</h3>" + "".join(
                    f"<a href='/probe?target={t}'>{t}</a> ({d.vendor} {d.host})<br>"
                    for t, d in collector.drivers.items())
                return self._send(200, body.encode(), "text/html; charset=utf-8")
            self._send(404, b"not found\n", "text/plain")

        def _send(self, code, body, ctype):
            self.send_response(code)
            self.send_header("Content-Type", ctype)
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)

        def log_message(self, *a):
            pass

    return H


def load(path):
    with open(path) as f:
        cfg = yaml.safe_load(f)
    defaults = cfg.get("defaults", {})
    drivers = {}
    for t in cfg.get("targets", []):
        t = {**defaults, **t}
        typ = t["type"].lower()
        if typ not in DRIVERS:
            sys.exit(f"target {t['name']}: type '{typ}' không hỗ trợ ({', '.join(DRIVERS)})")
        drivers[t["name"]] = DRIVERS[typ](t["name"], t)
    return cfg, drivers


def main():
    ap = argparse.ArgumentParser(description="Unified storage exporter (PowerVault / Unity / FlashSystem)")
    ap.add_argument("-c", "--config", default="/etc/storage_exporter/config.yml")
    ap.add_argument("--listen")
    ap.add_argument("--test", nargs="?", const="__all__", metavar="TARGET",
                    help="scrape 1 lần (tất cả hoặc 1 target), in metrics rồi thoát")
    ap.add_argument("-v", "--verbose", action="store_true")
    a = ap.parse_args()
    logging.basicConfig(level=logging.DEBUG if a.verbose else logging.INFO,
                        format="%(asctime)s %(levelname)s %(message)s")
    cfg, drivers = load(a.config)
    col = StorageCollector(drivers, cfg.get("workers", 5))
    if a.test:
        sys.stdout.write(render(col, None if a.test == "__all__" else a.test).decode())
        return
    listen = a.listen or cfg.get("listen", "0.0.0.0:9850")
    host, port = listen.rsplit(":", 1)
    log.info("storage exporter on %s - targets: %s", listen,
             ", ".join(f"{n}({d.vendor})" for n, d in drivers.items()))
    ThreadingHTTPServer((host, int(port)), make_handler(col)).serve_forever()


if __name__ == "__main__":
    main()
